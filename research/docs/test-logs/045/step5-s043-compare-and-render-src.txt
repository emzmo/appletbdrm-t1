/// - `service` → load the config; on `Err` (missing/zero-page) fall back to
///   `default_config()` with a loud `warn!`. A service must NOT crash-loop
///   on a missing config. This is the deliberate divergence from ADR-011
///   documented in ADR-019 D1.
/// - bare `tbard` (dev/real-non-service) → load the config; on `Err`
///   propagate `NoPagesConfigured` (exit 1) per ADR-011 — dev visibility,
///   no silent fallback.
///
/// Extracted from `real_main` so the policy is unit-testable without
/// starting the engine (which needs DRM master). The tests inject a
/// `ConfigManager` built with `with_paths` pointing at tempdirs.
fn resolve_config(
    cfg_mgr: &mut ConfigManager,
    dry_run: bool,
    service: bool,
) -> anyhow::Result<config::TouchbarConfig> {
    if dry_run {
        return Ok(demo_config());
    }
    match cfg_mgr.load() {
        Ok(c) => Ok(c),
        Err(e) => {
            if service {
                warn!(
                    "tbard: --service config load failed, using built-in default: {}",
                    e
                );
                Ok(default_config())
            } else {
                // ADR-011: surface the misconfiguration (dev visibility).
                log::error!("tbard: config load failed: {}", e);
                Err(e)
            }
        }
    }
}
